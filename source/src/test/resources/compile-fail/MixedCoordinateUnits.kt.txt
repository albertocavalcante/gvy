// Compile-fail proof for the typed coordinates in :source (issue #2391). This file must NOT
// compile: every statement in mixUnits mixes a unit, a base or an Int with a typed coordinate.
// The .txt suffix keeps it out of every compilation and formatter; it is documentation, not a test.
//
// Check it by compiling against the built :source classes as a separate module, so internal
// constructors are invisible as they are to every consumer. kotlinc compiles only .kt files:
//
//   ./gradlew :source:jar
//   dir="$(mktemp -d)"
//   cp source/src/test/resources/compile-fail/MixedCoordinateUnits.kt.txt "$dir/Mixed.kt"
//   cp source/src/test/resources/compile-fail/WellTypedCoordinates.kt.txt "$dir/Well.kt"
//   kotlinc -cp source/build/libs/source.jar -d "$dir/well" "$dir/Well.kt"   # must succeed
//   kotlinc -cp source/build/libs/source.jar -d "$dir/mixed" "$dir/Mixed.kt" # must fail
//   rm -rf "$dir"
//
// Expected: at least one error on each numbered line (line 5 reports both swapped arguments) and
// no error elsewhere. Last run with kotlinc 2.4.21: 13 errors, one or two per numbered line.

import com.github.albertocavalcante.gvy.source.CodePointColumn
import com.github.albertocavalcante.gvy.source.CodePointPosition
import com.github.albertocavalcante.gvy.source.Line
import com.github.albertocavalcante.gvy.source.Utf16Column
import com.github.albertocavalcante.gvy.source.Utf16LineIndex
import com.github.albertocavalcante.gvy.source.Utf16Position
import com.github.albertocavalcante.gvy.source.Utf16Span

fun mixUnits(index: Utf16LineIndex, line: Line, utf16: Utf16Column, codePoint: CodePointColumn) {
    val e1: CodePointColumn = utf16 // 1: a UTF-16 column is not a code-point column
    val e2: Utf16Column = codePoint // 2: nor the other way round
    CodePointPosition.of(line, utf16) // 3: a code-point position needs a code-point column
    Utf16Position.of(line, codePoint) // 4: a UTF-16 position needs a UTF-16 column
    Utf16Position.of(utf16, line) // 5: line and column cannot be swapped
    index.toCodePoint(CodePointPosition.of(line, codePoint)) // 6: no conversion from the unit it already has
    val e7: Line = 3 // 7: an Int is not a line; name its base with Line.ofZeroBased or Line.ofOneBased
    val e8: Utf16Column = Utf16Column(3) // 8: the constructor is internal; factories name the base
    val e9: Utf16Column = utf16 + 1 // 9: no arithmetic on columns, so no hand-written +1 or -1
    val e10: Int = line // 10: reading a line needs .zeroBased or .oneBased
    Utf16Span(Utf16Position.of(line, utf16), Utf16Position.of(line, utf16)) // 11: spans are built by checked factories
    val e12: CodePointColumn = utf16.toCodePoint() // 12: changing the column unit needs the line text
}

import com.github.albertocavalcante.gvy.source.DocumentVersion

fun matchKinds(version: DocumentVersion, other: DocumentVersion) {
    val c1: DocumentVersion = DocumentVersion.of(3)
    val c2: Int = version.value
    val c3 = version < other
    val c4 = version == other
    val c5: DocumentVersion? = DocumentVersion.ofOrNull(null)
    val c6: Boolean = version <= DocumentVersion.of(3)
    val c7: Long = version.value.toLong()
}

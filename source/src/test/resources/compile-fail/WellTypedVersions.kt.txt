    val c2: Int = version.value
    val c3 = version < other

    val c6: Boolean = version <= DocumentVersion.of(3)
    val c7: Long = version.value.toLong()
    val c8 = generation < generation.next()

    val c11: Long = generation.toLong()
    val c12 = epoch < epoch.next()

    val c14: OwnershipEpoch = epoch.next()
}
